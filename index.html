  <script>

    // Reset checklist daily

    window.onload = () => {

      const lastCheckedDate = localStorage.getItem('lastCheckedDate');

      const today = new Date().toLocaleDateString();if (lastCheckedDate !== today) {

    document.querySelectorAll('input[type="checkbox"]').forEach(checkbox => {

      checkbox.checked = false;

    });

    localStorage.setItem('lastCheckedDate', today);

  }

};

// Save checked state locally (optional, if you want to keep it during the day)

function handleCheck(e) {

  const task = e.target.id;

  localStorage.setItem(task, e.target.checked);

}
